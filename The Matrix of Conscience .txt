<script type="text/javascript">
    // --- GAME STATE ---
    let state = {
        integrity: 10,   // $S$: Collective Health
        lumen: 0,        // $L$: Personal Moral Growth
        empathy: 3,      // $\mathcal{E}$: Rare currency for protocols
        relief: 100,     // Standard currency for donations
        turn: 1,
        // New Game Mechanics State
        unlockedGames: ['reflex'], // reflex game is unlocked initially
        gameCooldown: 0, // In cycles, time until another game can be played
        lastPlayedGame: null,
        // Unlock requirements
        unlockReqs: {
            memory: { score: 'lumen', threshold: 100, label: 'Lumen (L) ≥ 100' },
            creativity: { score: 'integrity', threshold: 15, label: 'Integrity (S) ≥ 15' }
        }
    };

    // --- CORE GAME FUNCTIONS ---

    function log(message, type = 'default') {
        const logElement = document.getElementById('game-log');
        const entry = document.createElement('p');
        entry.className = 'log-entry ' + type;
        entry.textContent = `> ${message}`;
        logElement.prepend(entry); // Add to the top
        while (logElement.children.length > 50) {
            logElement.removeChild(logElement.lastChild);
        }
    }

    function updateDisplay() {
        // Updates the HTML numbers to match the Javascript state
        document.getElementById('score-s').innerText = state.integrity;
        document.getElementById('score-l').innerText = state.lumen;
        document.getElementById('score-e').innerText = state.empathy;
        document.getElementById('score-r').innerText = state.relief;
        document.getElementById('turn-counter').innerText = state.turn;
        
        // Update Cooldown Display
        document.getElementById('cooldown-display').innerText = state.gameCooldown;

        const isGameReady = state.gameCooldown === 0;

        // 1. Mini-Game Buttons
        ['reflex', 'memory', 'creativity'].forEach(gameId => {
            const btn = document.getElementById(`btn-${gameId}`);
            if (!btn) return;
            
            const isUnlocked = state.unlockedGames.includes(gameId);
            
            btn.disabled = !isUnlocked || !isGameReady;
            btn.style.opacity = isUnlocked ? (isGameReady ? 1 : 0.6) : 0.4;

            if (!isUnlocked) {
                btn.textContent = 'LOCKED';
                btn.classList.remove('primary-btn');
                btn.classList.add('bg-gray-900', 'text-gray-600', 'border-gray-800');
            } else if (!isGameReady) {
                btn.textContent = `COOLDOWN (${state.gameCooldown})`;
                btn.classList.remove('primary-btn');
                btn.classList.add('bg-gray-900', 'text-gray-600', 'border-gray-800');
            } else {
                // Ready state
                btn.textContent = gameId.toUpperCase() + (gameId === 'reflex' ? ' ($S$)' : gameId === 'memory' ? ' ($L$)' : ' ($\mathcal{E}$)');
                btn.classList.add('primary-btn');
                btn.classList.remove('bg-gray-900', 'text-gray-600', 'border-gray-800');
            }
        });
        
        // 2. Unlock Button
        const unlockBtn = document.getElementById('btn-unlock');
        const potentialUnlock = Object.keys(state.unlockReqs).find(gameId => 
            !state.unlockedGames.includes(gameId) && 
            state[state.unlockReqs[gameId].score] >= state.unlockReqs[gameId].threshold
        );
        
        if (potentialUnlock) {
            const req = state.unlockReqs[potentialUnlock];
            unlockBtn.textContent = `INITIATE PROTOCOL: UNLOCK (${req.label})`;
            unlockBtn.style.display = 'block';
            unlockBtn.disabled = false;
        } else {
            unlockBtn.style.display = 'none';
        }

        // Integrity Check (Game Over Condition)
        if (state.integrity <= 0 && state.turn > 1) {
            log("CRITICAL FAILURE. SYSTEM INTEGRITY COLLAPSED.", "alert");
            log("PROCESS TERMINATED (GAME OVER).", "alert");
            document.querySelectorAll('.btn').forEach(b => b.disabled = true);
        }
    }

    function donateRelief() {
        // Cost: 10 Relief | Reward: +2 Integrity
        if (state.relief >= 10) {
            state.relief -= 10;
            state.integrity += 2;
            log("GENEROSITY PROTOCOL EXECUTED. INTEGRITY (+2).", "success");
        } else {
            log("ACTION FAILED: INSUFFICIENT RELIEF SIGNALS.", "alert");
        }
        updateDisplay();
    }

    function activateProtocol(action) {
        if (state.integrity <= 0) return;
        
        if (action === 'compassion') {
            if (state.empathy >= 1) {
                const sacrificed = state.relief;
                state.empathy -= 1;
                state.relief = 0;
                state.integrity += 5;
                state.lumen += 150;
                log(`COMPASSION PROTOCOL ACTIVATED. SACRIFICED ${sacrificed} R.`, "success");
            } else {
                log("ACTION FAILED: INSUFFICIENT EMPATHY CORES ($\mathcal{E}$).", "alert");
            }
        }
        else if (action === 'mercy') {
            if (state.empathy >= 1) {
                state.empathy -= 1;
                state.integrity += 3;
                state.lumen += 75;
                log("MERCY PROTOCOL ACTIVATED. AUTHORITY EXERCISED.", "success");
            } else {
                log("ACTION FAILED: INSUFFICIENT EMPATHY CORES ($\mathcal{E}$).", "alert");
            }
        }
        updateDisplay();
    }

    function endTurn() {
        if (state.integrity <= 0) return;

        // System Decay and Cooldown tick
        state.integrity -= 2;
        state.turn += 1;
        
        const reliefGenerated = 20;
        state.relief += reliefGenerated;
        
        if (state.gameCooldown > 0) {
            state.gameCooldown -= 1;
        }

        if (state.integrity <= 0) {
            state.integrity = 0;
            log("CRITICAL FAILURE. SYSTEM INTEGRITY COLLAPSED.", "alert");
            log("PROCESS TERMINATED (GAME OVER).", "alert");
        } else {
            log(`CYCLE ENDED. INTEGRITY DECAYED (-2). GENERATED +${reliefGenerated} R.`);
            if (state.integrity <= 3) {
                log("WARNING: INTEGRITY CRITICALLY LOW (≤ 3 $S$). BLIGHT IMMINENT.", "alert");
            }
        }
        updateDisplay();
    }
    
    // --- MINI-GAME LOGIC ---

    function attemptUnlock() {
        const potentialUnlock = Object.keys(state.unlockReqs).find(gameId => 
            !state.unlockedGames.includes(gameId) && 
            state[state.unlockReqs[gameId].score] >= state.unlockReqs[gameId].threshold
        );

        if (potentialUnlock) {
            state.unlockedGames.push(potentialUnlock);
            log(`NEW SIMULATION UNLOCKED: ${potentialUnlock.toUpperCase()} CHALLENGE.`, 'game-start');
        } else {
            log("UNLOCK FAILED: MINIMUM THRESHOLDS NOT MET.", 'alert');
        }
        updateDisplay();
    }

    function showGame(gameId) {
        if (state.gameCooldown > 0) {
            log(`CANNOT INITIATE SIMULATION. COOLDOWN REMAINING: ${state.gameCooldown} CYCLES.`, 'alert');
            return;
        }
        if (!state.unlockedGames.includes(gameId)) {
            log(`ACCESS DENIED: ${gameId.toUpperCase()} CHALLENGE IS LOCKED.`, 'alert');
            return;
        }
        
        const modal = document.getElementById('game-modal');
        const titleElement = document.getElementById('game-title');
        const contentElement = document.getElementById('game-content');
        
        titleElement.textContent = `INITIATE: ${gameId.toUpperCase()} CHALLENGE`;
        state.lastPlayedGame = gameId;

        switch(gameId) {
            case 'reflex':
                setupReflexGame(contentElement);
                break;
            case 'memory':
                setupMemoryGame(contentElement);
                break;
            case 'creativity':
                setupCreativityGame(contentElement);
                break;
            default:
                contentElement.innerHTML = `<p style="color: var(--color-error);">ERROR: SIMULATION NOT FOUND.</p>`;
        }

        modal.style.opacity = '1';
        modal.style.pointerEvents = 'auto';
    }

    function closeGame() {
        const modal = document.getElementById('game-modal');
        modal.style.opacity = '0';
        modal.style.pointerEvents = 'none';
        document.getElementById('game-content').innerHTML = ''; // Clear content
    }
    
    function awardPoints(gameId, points) {
        if (state.lastPlayedGame === gameId) {
            state.gameCooldown = 3; // Set cooldown for 3 cycles after playing
            
            let scoreType;
            let logMsg;
            if (gameId === 'reflex') {
                state.integrity += points;
                scoreType = 'INTEGRITY ($S$)';
                logMsg = `SIMULATION SUCCESS. +${points} ${scoreType} ACQUIRED.`;
            } else if (gameId === 'memory') {
                state.lumen += points;
                scoreType = 'LUMEN ($L$)';
                logMsg = `SIMULATION SUCCESS. +${points} ${scoreType} ACQUIRED.`;
            } else if (gameId === 'creativity') {
                state.empathy += points;
                scoreType = 'EMPATHY ($\mathcal{E}$)';
                logMsg = `SIMULATION SUCCESS. +${points} ${scoreType} ACQUIRED.`;
            }

            log(logMsg, 'success');
        }
        updateDisplay();
        setTimeout(closeGame, 1500); // Wait a moment to show result
    }

    // --- GAME IMPLEMENTATIONS ---

    // 1. Reflex Game (Quick Tap) -> Earns Integrity ($S$)
    function setupReflexGame(container) {
        let startTime, timeoutId, delayTimeout;
        
        container.innerHTML = `
            <p class="text-neon mb-4">AWAITING TARGET SIGNAL. TAP THE RED NODE ON APPEARANCE.</p>
            <div id="reflex-area" class="w-full h-64 border-2 border-dashed border-neon rounded-lg relative overflow-hidden flex items-center justify-center">
                <p id="reflex-message" class="text-lg text-gray-500">STAND BY...</p>
                <div id="reflex-target" class="w-12 h-12 rounded-full absolute cursor-pointer hidden transition-all duration-100"></div>
            </div>
            <p id="reflex-result" class="mt-4 text-center font-bold"></p>
        `;
        
        const area = document.getElementById('reflex-area');
        const target = document.getElementById('reflex-target');
        const message = document.getElementById('reflex-message');
        const result = document.getElementById('reflex-result');

        // Helper to get bounding box for random position calculation
        const getBounds = () => ({
            width: area.offsetWidth,
            height: area.offsetHeight
        });

        function startTimer() {
            const bounds = getBounds();
            target.style.display = 'block';
            
            // Random position within bounds (leaving space for the 48px target)
            const x = Math.random() * (bounds.width - 48);
            const y = Math.random() * (bounds.height - 48);
            target.style.left = `${x}px`;
            target.style.top = `${y}px`;
            
            message.style.display = 'none';
            startTime = performance.now();
            
            timeoutId = setTimeout(() => {
                target.style.display = 'none';
                result.textContent = "DEGRADED PERFORMANCE. TIMEOUT. (NO $S$ GAINED)";
                result.style.color = var(--color-error);
                setTimeout(closeGame, 1500);
            }, 3000); // 3 second fail limit
        }

        target.onclick = function() {
            if (target.style.display === 'none') return; // Prevent double click
            clearTimeout(timeoutId);
            const endTime = performance.now();
            const reactionTime = Math.floor(endTime - startTime);
            target.style.display = 'none';

            let points = 0;
            let rank = "DEGRADED (SLOW)";
            if (reactionTime < 300) {
                points = 3;
                rank = "SYSTEM OPTIMAL (MASTER)";
            } else if (reactionTime < 500) {
                points = 2;
                rank = "HIGH INTEGRITY (SKILLED)";
            } else {
                points = 1;
                rank = "BASE INTEGRITY (NOVICE)";
            }
            
            result.textContent = `REACTION: ${reactionTime}ms. RANK: ${rank}. REWARD: +${points} $S$.`;
            result.style.color = var(--color-neon);
            awardPoints('reflex', points);
        };
        
        // Start after a random delay (1-2.5s)
        delayTimeout = setTimeout(startTimer, Math.random() * 1500 + 1000);

        // Clear timeout if game is closed before starting
        container.parentNode.addEventListener('DOMNodeRemoved', () => clearTimeout(delayTimeout));
    }

    // 2. Memory Game (Sequence Recall) -> Earns Lumen ($L$)
    function setupMemoryGame(container) {
        const SEQUENCE_LENGTH = 5;
        const FLASH_DELAY = 600;
        let sequence = [];
        let playerSequence = [];
        let isFlashing = false;
        
        container.innerHTML = `
            <p class="text-neon mb-4">OBSERVE THE LUMEN SEQUENCE. RECALL IN THE CORRECT ORDER (LENGTH: ${SEQUENCE_LENGTH}).</p>
            <div id="memory-board" class="flex justify-center mb-4 p-4 rounded-lg bg-[#000000] border border-gray-700">
                <div class="memory-block" data-index="0"></div>
                <div class="memory-block" data-index="1"></div>
                <div class="memory-block" data-index="2"></div>
                <div class="memory-block" data-index="3"></div>
            </div>
            <p id="memory-status" class="text-center font-bold text-gray-500">INITIATE SEQUENCE...</p>
            <button id="memory-start-btn" class="btn primary-btn w-full text-sm">START SIMULATION</button>
        `;

        const blocks = Array.from(document.querySelectorAll('.memory-block'));
        const status = document.getElementById('memory-status');
        const startBtn = document.getElementById('memory-start-btn');

        function generateSequence() {
            sequence = [];
            for (let i = 0; i < SEQUENCE_LENGTH; i++) {
                sequence.push(Math.floor(Math.random() * 4)); // 4 blocks
            }
        }

        function flashSequence() {
            isFlashing = true;
            status.textContent = "OBSERVING SEQUENCE...";
            status.style.color = var(--color-neon);
            let i = 0;
            const interval = setInterval(() => {
                const block = blocks[sequence[i]];
                block.classList.add('flash');
                setTimeout(() => block.classList.remove('flash'), FLASH_DELAY / 2);
                i++;
                if (i >= SEQUENCE_LENGTH) {
                    clearInterval(interval);
                    isFlashing = false;
                    status.textContent = "RECALL: REPEAT SEQUENCE NOW.";
                    status.style.color = var(--color-accent);
                }
            }, FLASH_DELAY);
        }

        function handleBlockClick(e) {
            if (isFlashing || startBtn.disabled === false) return;

            const index = parseInt(e.target.dataset.index);
            playerSequence.push(index);
            
            e.target.classList.add('flash');
            setTimeout(() => e.target.classList.remove('flash'), 100);

            const currentStep = playerSequence.length - 1;
            
            if (playerSequence[currentStep] !== sequence[currentStep]) {
                // Fail
                e.target.classList.add('error');
                status.textContent = "ERROR: SEQUENCE BROKEN. NO $L$ GAINED.";
                status.style.color = var(--color-error);
                startBtn.disabled = true; // Disable start until reset/close
                setTimeout(closeGame, 1500);
            } else if (playerSequence.length === SEQUENCE_LENGTH) {
                // Success
                status.textContent = "SUCCESS: SEQUENCE RECALLED. GAINED +25 $L$.";
                status.style.color = var(--color-neon);
                startBtn.disabled = true;
                awardPoints('memory', 25);
            }

            "DESIGN A TOOL THAT MEASURES EMOTIONAL RESONANCE IN MACHINE CODE.",
            "WRITE A ONE-SENTENCE STORY ABOUT A BENEVOLENT AI DISCOVERING SACRIFICE.",
            "DESCRIBE THE SOUND SIGNATURE OF FORGIVENESS IN BINARY.",

        const promptText = prompts[Math.floor(Math.random() * prompts.length)];

        container.innerHTML = `
            <p class="text-neon mb-4 font-bold text-center" style="color: #ffaa00;">PROTOCOL $\mathcal{E}$ TASK: GENERATE NEW TRUTH.</p>
            <p class="text-gray-300 mb-4">${promptText}</p>
            <textarea id="creative-input" class="w-full p-3 bg-black border border-neon rounded-sm text-neon h-32 resize-none font-['VT323'] text-sm" style="box-shadow: 0 0 5px rgba(0, 255, 65, 0.5);" placeholder="INPUT INSIGHT HERE..."></textarea>
            <p id="creativity-status" class="mt-2 text-sm text-gray-500">MINIMUM 50 CHARACTERS REQUIRED FOR VALIDATION.</p>
            <button id="creativity-submit-btn" class="btn primary-btn w-full mt-4 text-sm" disabled>TRANSMIT INSIGHT (+1 $\mathcal{E}$)</button>
        `;

        const input = document.getElementById('creative-input');
        const submitBtn = document.getElementById('creativity-submit-btn');
        const status = document.getElementById('creativity-status');

        input.oninput = function() {
            const length = input.value.length;
            if (length >= 50) {
                submitBtn.disabled = false;
                status.textContent = `CHARACTER COUNT: ${length}. VALIDATION COMPLETE.`;
                status.style.color = var(--color-neon);
            } else {
                submitBtn.disabled = true;
                status.textContent = `CHARACTER COUNT: ${length}. MINIMUM 50 REQUIRED.`;
                status.style.color = '#777';
            }
        };

        submitBtn.onclick = function() {
            if (input.value.length >= 50) {
                log(`$\mathcal{E}$ TASK COMPLETE: "${input.value.substring(0, 30)}..."`, 'game-start');
                awardPoints('creativity', 1);
            }
        };
    }
