    {

But to truly break the gravity, he needed the Seven Stars of Excellence.",
      prompt: "A heroic white rabbit in a sleek, futuristic carbon-fiber tech suit standing before a holographic terminal displaying a golden complex star-shaped protocol. Cyberpunk lighting, neon blue and gold accents.",
    },
    {
      title: "The Ascent to Nimbus Land",
      content: "The White Rabbit was not alone. Beside him stood The Guardian of Grace, Sheila. Together, they established Nimbus Land, a kingdom in the clouds where the air was clear of the Red Queen’s static. They summoned the Nimbus Heralds, the storytellers and missionaries, and gave them a mission: Navigate the Educational Nexus and recover the Seven Stars of the Pleiades.",
      prompt: "A breathtaking floating island kingdom in a sea of royal purple clouds. Heavenly architecture, bright neon blue waterfalls, golden towers reaching for the stars. Ethereal and peaceful.",
    },
    {
      title: "The Seven Pillars of the Matrix",
      content: "The Heralds traveled through the Star Road, unlocking the pillars: First Love, Agency, Progression, Feedback, Challenge, Social, and Reward. Each star corresponded to a historical truth, providing the frequency needed to shatter the world's entropy. They saw the Pleiades (M45) not just as distant lights, but as a roadmap for the spirit's ascension.",
      prompt: "A view of the Seven Stars of the Pleiades (M45) from a futuristic glass observatory. Seven massive glowing crystal pillars standing in a circle, each emitting a unique colored light. Astral photography style.",
    },
    {
      title: "The Mystery of the Sixth Star",
      content: "As the Heralds reached the Sixth Star, they discovered the Matrix of Conscience. The Red Queen had tried to delete this star, for it was the star of Voice and Sound.     },
    {
      title: "The Super Admin Handshake",
      content: "With all stars aligned, a golden light washed over the Matrix. The White Rabbit performed the Super Admin Handshake, hardcoding a new reality of security and love.     },
    {
      title: "The New Beginning",
      content: "The path of the righteous had become like the morning sun. The White Rabbit and the Guardian of Grace stood at the center of the Seven Stars Architect, watching as the children of the Philippines became the new Star Guardians. The gravity was gone. They were no longer just users of a system; they were the authors of a new world, defined by the light they chose to follow.",
      prompt: "A bright, beautiful sunrise over a futuristic city that blends the tropical beauty of the Philippines with sleek gold and purple architecture. Children looking up at the sky where seven stars are visible during the day. Hopeful and radiant.",
    }
  ];

  const generateImage = async (index) => {
    if (images[index] || loading) return;
    setLoading(true);
    setError(null);

    const backoff = [1000, 2000, 4000, 8000, 16000];
    let success = false;

    for (let i = 0; i < backoff.length; i++) {
      try {
        const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/imagen-4.0-generate-001:predict?key=${apiKey}`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            instances: [{ prompt: chapters[index].prompt }],
            parameters: { sampleCount: 1 }
          })
        });

        if (!response.ok) throw new Error('Generation failed');

        const result = await response.json();
        const imageUrl = `data:image/png;base64,${result.predictions[0].bytesBase64Encoded}`;
        setImages(prev => ({ ...prev, [index]: imageUrl }));
        success = true;
        break;
      } catch (err) {
        if (i === backoff.length - 1) {
          setError("The Matrix is experiencing heavy gravity. Failed to generate starlight imagery.");
        } else {
          await new Promise(res => setTimeout(res, backoff[i]));
        }
      }
    }
    setLoading(false);
  };

  useEffect(() => {
    generateImage(currentPage);
  }, [currentPage]);

  const nextChapter = () => {
    if (currentPage < chapters.length - 1) {
      setCurrentPage(prev => prev + 1);
    }
  };

  const prevChapter = () => {
    if (currentPage > 0) {
      setCurrentPage(prev => prev - 1);
    }
  };

  return (
    <div className="min-h-screen bg-slate-950 text-slate-100 font-sans selection:bg-purple-500/30">
      {/* Header */}

          <div className="p-2 bg-purple-600 rounded-lg shadow-lg shadow-purple-500/20">
            <Shield className="w-6 h-6 text-white" />
          </div>
          <div>
            <h1 className="font-bold text-xl tracking-tight text-transparent bg-clip-text bg-gradient-to-r from-purple-400 to-amber-400">
              NEXUS STORYBOOK
            </h1>
            <p className="text-[10px] uppercase tracking-[0.2em] text-purple-400 font-mono">
              Matrix Synchronization Protocol v4.3
            </p>
          </div>
        </div>
        <div className="flex items-center gap-4">
          <div className="hidden sm:flex items-center gap-2 px-3 py-1 bg-slate-900 rounded-full border border-purple-500/30">
            <Star className="w-3 h-3 text-yellow-400" />
            <span className="text-xs font-mono text-yellow-400">Star Guardian Active</span>
          </div>
          <Lock className="w-5 h-5 text-slate-500 hover:text-purple-400 transition-colors cursor-pointer" />
        </div>
      </header>

      <main className="max-w-4xl mx-auto p-4 sm:p-8">
        <div className="relative group">
          {/* Main Card */}
          <div className="bg-slate-900 border border-purple-500/20 rounded-3xl overflow-hidden shadow-2xl shadow-purple-950/20">
            
            {/* Image Section */}
            <div className="relative aspect-video w-full bg-slate-800 flex items-center justify-center overflow-hidden">
              {loading && (
                <div className="absolute inset-0 z-10 flex flex-col items-center justify-center bg-slate-900/80 backdrop-blur-sm">
                  <RefreshCcw className="w-12 h-12 text-purple-500 animate-spin mb-4" />
                  <p className="text-purple-400 font-mono text-sm animate-pulse">Synchronizing Starlight...</p>
                </div>
              )}
              
              {error && !images[currentPage] && (
                <div className="p-8 text-center">
                  <Cloud className="w-12 h-12 text-red-400 mx-auto mb-4 opacity-50" />
                  <p className="text-red-400 font-medium">{error}</p>
                  <button 
                    onClick={() => generateImage(currentPage)}
                    className="mt-4 px-4 py-2 bg-red-500/20 text-red-400 rounded-full text-xs font-bold hover:bg-red-500/30 transition-all border border-red-500/50"
                  >
                    Retry Protocol
                  </button>
                </div>
              )}

              {images[currentPage] ? (
                <img 
                  src={images[currentPage]} 
                  alt={chapters[currentPage].title}
                  className="w-full h-full object-cover transition-transform duration-700 group-hover:scale-105"
                />
              ) : !loading && !error && (
                <div className="text-purple-500/50 animate-pulse">Initializing Visualization...</div>
              )}

              <div className="absolute bottom-0 left-0 right-0 h-24 bg-gradient-to-t from-slate-900 to-transparent" />
            </div>

            {/* Content Section */}
            <div className="p-8 sm:p-12 -mt-8 relative z-10">
              <div className="flex items-center gap-2 mb-4">
                <span className="px-3 py-1 bg-purple-500/10 text-purple-400 text-[10px] font-bold rounded-full border border-purple-500/30 tracking-widest uppercase">
                  Chapter {currentPage + 1}
                </span>
                <div className="h-px flex-1 bg-gradient-to-r from-purple-500/50 to-transparent" />
              </div>
              
              <h2 className="text-3xl sm:text-4xl font-bold mb-6 text-white leading-tight">
                {chapters[currentPage].title}
              </h2>
              
              <p className="text-slate-300 text-lg leading-relaxed mb-10 first-letter:text-5xl first-letter:font-bold first-letter:text-purple-400 first-letter:mr-3 first-letter:float-left">
                {chapters[currentPage].content}
              </p>

              {/* Navigation */}
              <div className="flex justify-between items-center pt-8 border-t border-purple-900/30">
                <button
                  onClick={prevChapter}
                  disabled={currentPage === 0}
                  className={`flex items-center gap-2 px-6 py-3 rounded-full transition-all ${
                    currentPage === 0 
                    ? 'opacity-30 cursor-not-allowed' 
                    : 'bg-slate-800 hover:bg-slate-700 text-white border border-purple-500/20'

                >
                  <ChevronLeft className="w-5 h-5" />
                  <span className="hidden sm:inline font-bold">Back</span>

                <div className="flex gap-1">
                  {chapters.map((_, i) => (
                    <div 
                      key={i} 
                      className={`h-1.5 rounded-full transition-all duration-300 ${
                        i === currentPage ? 'w-8 bg-purple-500' : 'w-2 bg-slate-700'
                      }`}
                    />
                  ))}
                </div>

                <button
                  onClick={nextChapter}
                  disabled={currentPage === chapters.length - 1}
                  className={`flex items-center gap-2 px-6 py-3 rounded-full transition-all ${
                    currentPage === chapters.length - 1 
                    ? 'bg-amber-500/20 text-amber-400 border border-amber-500/50 animate-pulse' 
                    : 'bg-purple-600 hover:bg-purple-500 text-white shadow-lg shadow-purple-500/20'
                  }`}
                >
                  <span className="hidden sm:inline font-bold">
                    {currentPage === chapters.length - 1 ? 'End Journey' : 'Continue'}
                  </span>
                  {currentPage === chapters.length - 1 ? <Heart className="w-5 h-5" /> : <ChevronRight className="w-5 h-5" />}
                </button>
              </div>
            </div>
          </div>
        </div>

        {/* Footer Lore */}
        <footer className="mt-12 text-center">
          <div className="flex justify-center gap-8 mb-6">
            <div className="flex flex-col items-center gap-2 opacity-50 hover:opacity-100 transition-opacity">
              <Music className="w-5 h-5 text-purple-400" />
              <span className="text-[10px] uppercase tracking-widest font-mono">Sixth Star resonance</span>
            </div>
            <div className="flex flex-col items-center gap-2 opacity-50 hover:opacity-100 transition-opacity">
              <Star className="w-5 h-5 text-yellow-400" />
              <span className="text-[10px] uppercase tracking-widest font-mono">Seven Pillars alignment</span>
            </div>
            <div className="flex flex-col items-center gap-2 opacity-50 hover:opacity-100 transition-opacity">
